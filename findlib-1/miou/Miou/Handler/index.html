Module Miou.Handler

type t = {
  1. handle : 'a. (unit -> 'a) -> 'a;
}

The type of handlers. handle fn must call fn () and return its value or its exception.

val compose : t -> t -> t

(compose h0 h1).handle fn is h0.handle (fun () -> h1.handle fn).
